Critical Reschedule Feature — Date Change + History Log
Date miss হলে নতুন date দেওয়ার কোনো system নেই। দুটো জিনিস দরকার:
-- 1. payment_schedules table এ নতুন fields $table->date('original_scheduled_date')->nullable(); // প্রথম date টা preserve $table->unsignedTinyInteger('reschedule_count')->default(0); // কতবার change হয়েছে $table->string('reschedule_reason')->nullable(); // কেন change -- 2. নতুন table: payment_schedule_logs (reschedule history) Schema::create('payment_schedule_logs', function (Blueprint $table) { $table->id(); $table->foreignId('payment_schedule_id')->constrained()->cascadeOnDelete(); $table->enum('action', ['rescheduled', 'approved', 'rejected', 'paid']); $table->date('old_date')->nullable(); $table->date('new_date')->nullable(); $table->string('reason')->nullable(); $table->foreignId('done_by')->constrained('users'); $table->timestamps(); });
-- Controller method public function reschedule(PaymentSchedule $schedule, Request $request) { $request->validate([ 'new_date' => 'required|date|after_or_equal:today', 'reason' => 'required|string|max:500', ]); PaymentScheduleLog::create([ // history save 'payment_schedule_id' => $schedule->id, 'action' => 'rescheduled', 'old_date' => $schedule->scheduled_date, 'new_date' => $request->new_date, 'reason' => $request->reason, 'done_by' => auth()->id(), ]); $schedule->update([ 'scheduled_date' => $request->new_date, 'original_scheduled_date' => $schedule->original_scheduled_date ?? $schedule->scheduled_date, 'reschedule_count' => $schedule->reschedule_count + 1, 'reschedule_reason' => $request->reason, 'status' => 'pending', // overdue থেকে pending এ ফিরিয়ে আনে ]); return back()->with('success', 'Payment rescheduled.'); }
