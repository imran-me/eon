// POST /purchases/make/payment/{id}

public function make_payment(Request $request, $role, $id)
{
    $validated = $request->validate([
        'payment_method' => 'required|in:cash,card,advance,checque,bank_transfer,mobile_banking,other',
        'bank_id'        => 'required|exists:banks,id',
        'payment_date'   => 'required|date',
        'reference_no'   => 'nullable|string|max:255',
        'payment_amount' => 'required|numeric|min:0.01',
        'schedule_id'    => 'nullable|exists:payment_schedules,id',
    ]);

    $purchase = Purchase::findOrFail($id);

    if ($validated['payment_amount'] > $purchase->due_amount) {
        return redirect()->back()->with('error', 'Payment amount cannot exceed due amount!');
    }

    try {
        DB::transaction(function () use ($validated, $purchase) {

            $acc = Bank::find($validated['bank_id']);
            $opening_balance = $acc?->balance ?? 0;

            // 1. Transaction record (money going OUT)
            Transaction::create([
                'user_id'        => $purchase->supplier_id,
                'user_type'      => 'supplier',
                'type'           => 'purchase',
                'account_id'     => $validated['bank_id'],
                'payment_date'   => $validated['payment_date'],
                'reference_no'   => $validated['reference_no'],
                'payment_method' => $validated['payment_method'],
                'invoice_id'     => $purchase->id,
                'old_balance'    => $opening_balance,
                'debit'          => $validated['payment_amount'],
                'credit'         => 0,
                'balance'        => $opening_balance - $validated['payment_amount'],
                'note'           => 'Payment from Make Payment.',
            ]);

            // 2. Bank balance update (money going out → decrement)
            if ($acc) {
                $acc->update(['balance' => $opening_balance - $validated['payment_amount']]);
            }

            // 3. Purchase update
            $newPaidTotal = min($purchase->paid_amount + $validated['payment_amount'], $purchase->total_amount);
            $newDueAmount = max($purchase->due_amount - $validated['payment_amount'], 0);
            $purchase->update([
                'paid_amount'    => $newPaidTotal,
                'due_amount'     => $newDueAmount,
                'payment_status' => $newDueAmount <= 0 ? 'paid' : 'partial',
            ]);

            // 4. Journal Entry: Debit AP 2110 / Credit Bank
            $apAccount   = \App\Models\Account::where('code', config('accounts.accounts_payable'))->firstOrFail();
            $bank        = Bank::find($validated['bank_id']);
            if (!$bank || !$bank->account_id) {
                throw new \Exception('Bank is not linked to a chart-of-accounts account.');
            }

            $journal = \App\Models\JournalEntry::create([
                'company_id'  => auth()->user()->company_id ?? 2,
                'created_by'  => auth()->id(),
                'date'        => $validated['payment_date'],
                'reference'   => $validated['reference_no'] ?? $purchase->purchase_no,
                'source'      => 'purchase',
                'source_id'   => $purchase->id,
                'description' => 'Payment made — ' . $purchase->purchase_no,
            ]);

            // Debit: AP account (reducing payable)
            \App\Models\JournalItem::create([
                'journal_entry_id' => $journal->id,
                'account_id'       => $apAccount->id,
                'debit'            => $validated['payment_amount'],
                'credit'           => 0,
                'note'             => 'Payable cleared — ' . $purchase->purchase_no,
            ]);

            // Credit: Bank account (money leaving)
            \App\Models\JournalItem::create([
                'journal_entry_id' => $journal->id,
                'account_id'       => $bank->account_id,
                'debit'            => 0,
                'credit'           => $validated['payment_amount'],
                'note'             => 'Payment sent',
            ]);

            // 5. Schedule mark paid
            if (!empty($validated['schedule_id'])) {
                \App\Models\PaymentSchedule::where('id', $validated['schedule_id'])
                    ->where('schedulable_type', \App\Models\Purchase::class)
                    ->where('schedulable_id', $purchase->id)
                    ->update(['status' => 'paid', 'paid_date' => $validated['payment_date']]);
            }
        });
    } catch (\Throwable $th) {
        DB::rollBack();
        return redirect()->back()->with('error', $th->getMessage());
    }

    return redirect()->back()->with('success', 'Payment of ৳' . number_format($validated['payment_amount'], 2) . ' sent successfully!');
}

public function make_payment(Request $request, $role, $id)
{
    $validated = $request->validate([
        'payment_method' => 'required|string',
        'bank_id'        => 'required|exists:banks,id',
        'payment_date'   => 'required|date',
        'reference_no'   => 'nullable|string|max:255',
        'payment_amount' => 'required|numeric|min:0.01',
        'schedule_id'    => 'nullable|exists:payment_schedules,id',
    ]);

    $ticketSale = TicketSale::findOrFail($id);

    if ($validated['payment_amount'] > $ticketSale->due_amount) {
        return redirect()->back()->with('error', 'Payment amount cannot exceed due amount!');
    }

    DB::transaction(function () use ($validated, $ticketSale) {
        $this->manageBankPayment(
            (object) ['bank_id' => $validated['bank_id'], 'payment_method' => $validated['payment_method'], 'payment_date' => $validated['payment_date'], 'reference_no' => $validated['reference_no']],
            $ticketSale->id,
            $validated['payment_amount']
        );

        $newPaid = min($ticketSale->paid_amount + $validated['payment_amount'], $ticketSale->total_amount);
        $newDue  = max($ticketSale->due_amount  - $validated['payment_amount'], 0);
        $ticketSale->update([
            'paid_amount'    => $newPaid,
            'due_amount'     => $newDue,
            'payment_status' => $newDue <= 0 ? 'paid' : 'partial',
        ]);

        // Journal: Debit Bank / Credit AR 1130
        $arAccount = \App\Models\Account::where('code', config('accounts.accounts_receivable'))->firstOrFail();
        $bank      = Bank::findOrFail($validated['bank_id']);
        $journal   = \App\Models\JournalEntry::create([
            'company_id'  => auth()->user()->company_id ?? 2,
            'created_by'  => auth()->id(),
            'date'        => $validated['payment_date'],
            'reference'   => $ticketSale->invoice_no,
            'source'      => 'ticket_sale',
            'source_id'   => $ticketSale->id,
            'description' => 'Ticket sale payment received — ' . $ticketSale->invoice_no,
        ]);
        \App\Models\JournalItem::create(['journal_entry_id' => $journal->id, 'account_id' => $bank->account_id, 'debit' => $validated['payment_amount'], 'credit' => 0, 'note' => 'Payment received']);
        \App\Models\JournalItem::create(['journal_entry_id' => $journal->id, 'account_id' => $arAccount->id, 'debit' => 0, 'credit' => $validated['payment_amount'], 'note' => 'AR cleared']);

        if (!empty($validated['schedule_id'])) {
            \App\Models\PaymentSchedule::where('id', $validated['schedule_id'])
                ->update(['status' => 'paid', 'paid_date' => $validated['payment_date']]);
        }
    });

    return redirect()->back()->with('success', 'Payment recorded!');
}

class PaymentScheduleController extends Controller
{
    // GET /payment-schedules?type=receive|pay&status=pending|overdue&date_from=&date_to=
    public function index(Request $request)
    {
        $query = PaymentSchedule::with('schedulable')
            ->when($request->type, fn($q) => $q->where('type', $request->type))
            ->when($request->status, fn($q) => $q->where('status', $request->status))
            ->when($request->date_from, fn($q) => $q->where('scheduled_date', '>=', $request->date_from))
            ->when($request->date_to, fn($q) => $q->where('scheduled_date', '<=', $request->date_to))
            ->orderBy('scheduled_date');

        $schedules  = $query->paginate(20);
        $todayCount = PaymentSchedule::where('scheduled_date', today())->where('status', 'pending')->count();
        $overdueAmt = PaymentSchedule::where('status', 'overdue')->sum('amount');

        return view('payment-schedules.index', compact('schedules', 'todayCount', 'overdueAmt'));
    }

    // POST /payment-schedules — Schedule তৈরি করো
    public function store(Request $request)
    {
        $request->validate([
            'schedulable_type' => 'required|in:sale,purchase,ticket_sale,ticket_purchase',
            'schedulable_id'   => 'required|integer',
            'schedules'        => 'required|array|min:1',
            'schedules.*.amount' => 'required|numeric|min:0.01',
            'schedules.*.date'   => 'required|date',
        ]);

        // Model resolve
        $modelMap = [
            'sale'             => \App\Models\Sale::class,
            'purchase'         => \App\Models\Purchase::class,
            'ticket_sale'      => \App\Models\TicketSale::class,
            'ticket_purchase'  => \App\Models\TicketPurchase::class,
        ];
        $model = $modelMap[$request->schedulable_type]::findOrFail($request->schedulable_id);

        // Total scheduled ≤ due_amount
        $alreadyScheduled = PaymentSchedule::where('schedulable_type', get_class($model))
            ->where('schedulable_id', $model->id)
            ->whereIn('status', ['pending', 'overdue'])
            ->sum('amount');

        $newTotal = collect($request->schedules)->sum('amount');
        if (($alreadyScheduled + $newTotal) > $model->due_amount) {
            return back()->withErrors(['amount' => 'Schedule total exceeds due amount: ' . $model->due_amount]);
        }

        // Party info
        $partyInfo = $this->resolveParty($request->schedulable_type, $model);
        $type = in_array($request->schedulable_type, ['sale', 'ticket_sale']) ? 'receive' : 'pay';

        foreach ($request->schedules as $sched) {
            PaymentSchedule::create([
                'schedulable_type' => get_class($model),
                'schedulable_id'   => $model->id,
                'type'             => $type,
                'party_type'       => $partyInfo['type'],
                'party_id'         => $partyInfo['id'],
                'amount'           => $sched['amount'],
                'scheduled_date'   => $sched['date'],
                'status'           => 'pending',
                'note'             => $sched['note'] ?? null,
                'created_by'       => auth()->id(),
            ]);
        }

        return back()->with('success', count($request->schedules) . ' schedule(s) created.');
    }

    private function resolveParty(string $type, $model): array
    {
        return match($type) {
            'sale'            => ['type' => 'customer', 'id' => $model->customer_id],
            'purchase'        => ['type' => 'vendor',   'id' => $model->supplier_id],
            'ticket_sale'     => ['type' => 'agent',    'id' => $model->agent_id],
            'ticket_purchase' => ['type' => 'vendor',   'id' => $model->vendor_id],
        };
    }
}
